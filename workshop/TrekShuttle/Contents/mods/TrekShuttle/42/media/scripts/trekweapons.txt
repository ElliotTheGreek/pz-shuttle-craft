/*  Weapon models -- and they live in `module Base`, deliberately.

    The first attempt declared this block in `module TrekShuttle` alongside the
    hull and the helm, where every other model in this mod lives, and the
    bat'leth drew nothing in hand. The log said:

        WARN MeshAssetManager.loadCallback
             Failed to load asset: AssetPath{ "TrekBatlethModel" }

    which is the tell: the engine is trying to load the *model block's name* as
    if it were a mesh path. It only does that when the name resolved to no
    ModelScript at all -- had the block been found, the asset it failed on
    would have been `weapons/2handed/TREK_Batleth`.

    So `WeaponSprite` does not resolve the way `StaticModel` does.
    ScriptManager.resolveModelScript takes a ScriptModule, and the hull and
    helm prove an in-module lookup works for StaticModel; the weapon path
    evidently searches Base. Every vanilla weapon model is in `module Base`
    (media/scripts/generated/models_weapons.txt), and the one Workshop mod
    shipping custom in-hand blades puts its models in `module Base` while
    keeping its items in its own module. Both agree, so this follows them.

    A separate file because no vanilla script file declares two modules, and
    because this is what vanilla does: models_weapons.txt beside weapon.txt.

    The item that uses this stays in module TrekShuttle, in trekshuttle.txt,
    and refers to it by bare name.

    `Bip01_Prop2` (the hand) and `world` (the ground) are optional and left
    out: their offsets can only honestly be set by looking at the weapon in a
    fist. Vanilla's Katana is the baseline to copy from when that happens.

    The bat'leth's two back attachments are not optional. An item on the back
    is drawn at bone x the body's attachment x the item's own attachment of
    the same name. The back slots hang a weapon by its Y, which is the
    bat'leth's length since it is held by one end, but its crescent bows
    along X and would stand out from the back without them (when it was held
    by the middle its length was X, and it stood across the shoulders like
    horns). tools/gen_batleth.py writes that block, from the mesh it builds;
    do not edit it here.  */

module Base
{
    model TrekBatlethModel
    {
        mesh = weapons/2handed/TREK_Batleth,
        texture = weapons/2handed/TREK_Batleth,
        attachment blade_back
        {
            offset = 0.02 -0.0073 0.0329,
            rotate = 0.0 90.0 0.0,
        }
        attachment big_blade_back_bag
        {
            offset = 0.12 -0.0436 0.0506,
            rotate = 25.0 90.0 0.0,
        }
    }

    model TrekMeklethModel
    {
        mesh = weapons/1handed/TREK_Mekleth,
        texture = weapons/1handed/TREK_Mekleth,
    }

    model TrekLirpaModel
    {
        mesh = weapons/2handed/TREK_Lirpa,
        texture = weapons/2handed/TREK_Lirpa,
    }

    model TrekUshaanTorModel
    {
        mesh = weapons/1handed/TREK_UshaanTor,
        texture = weapons/1handed/TREK_UshaanTor,
    }

    /*  The phaser (PHASERS.md 4). Unlike the blades, this one carries
        attachments, and they are not guesses: tools/bake_phaser.py puts the
        mesh in the frame of vanilla's M9 (the mesh behind Handgun03) with its
        grip where the M9's grip is, so `Bip01_Prop2` and `world` are
        Handgun03's own, copied. The muzzle is the phaser's emitter, as the
        bake measured it; the Handgun animation set reads it for the flash.
        If it sits wrong in the fist, the grip fit in the bake is the thing
        to move, not these numbers.  */
    model TrekPhaserModel
    {
        mesh = weapons/firearm/TREK_Phaser,
        texture = weapons/firearm/TREK_Phaser,
        attachment muzzle
        {
            offset = 0.0 0.1656 -0.0372,
            rotate = -90.0 0.0 0.0,
        }
        attachment world
        {
            offset = -0.009 0.062 0.0,
            rotate = 0.0 0.0 0.0,
        }
        attachment Bip01_Prop2
        {
            offset = -0.0338 0.0424 0.0,
            rotate = 90.0008 1.1964 -90.0396,
        }
    }
}
