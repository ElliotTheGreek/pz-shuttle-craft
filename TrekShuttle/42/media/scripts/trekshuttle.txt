module TrekShuttle
{
    /*  The hull and the helm are world models rather than tile sprites. A
        custom tile would need a texture pack built with TileZed; a model is a
        mesh plus a PNG, and can be dropped on any square.

        The meshes live in media/models_X and the textures in media/textures,
        both generated by tools/gen_shuttle.py and tools/gen_helm.py.

        scale is the one number to touch if either reads too large or too
        small in game: the meshes are authored at 1 unit == 1 tile. The hull is
        drawn three tiles wide and five long, which is the footprint
        C.Footprint in TREK_Config.lua asks the world for before it will set
        the ship down. Nothing in the engine ties those together, so change
        both or neither.  */

    model TrekShuttleHullModel
    {
        mesh = TREK_Shuttle,
        texture = TREK_Shuttle,
        scale = 1.0,
    }

    model TrekHelmConsoleModel
    {
        mesh = TREK_Helm,
        texture = TREK_Helm,
        scale = 1.0,
    }

    item TrekShuttleHull
    {
        ItemType = base:normal,
        Type = Normal,
        DisplayName = Shuttlecraft,
        DisplayCategory = Furniture,
        Icon = TREK_Shuttle,
        Weight = 90.0,
        WorldStaticModel = TrekShuttleHullModel,
        StaticModel = TrekShuttleHullModel,
        Tooltip = Tooltip_TREK_Shuttle,
    }

    item TrekHelmConsole
    {
        ItemType = base:normal,
        Type = Normal,
        DisplayName = Helm Console,
        DisplayCategory = Furniture,
        Icon = TREK_Shuttle,
        Weight = 70.0,
        WorldStaticModel = TrekHelmConsoleModel,
        StaticModel = TrekHelmConsoleModel,
    }

    sound TREK_PhaserPulse
    {
        category = Weapon,
        loop = false,
        is3D = true,
        clip
        {
            file = media/sound/TREK_PhaserPulse.wav,
            distanceMin = 2,
            distanceMax = 35,
            volume = 0.85,
        }
    }

    /*  The phaser.

        An ordinary build 42 sidearm in every respect except that TREK_Phaser
        .lua keeps it charged, unjammed and undamaged, which is what makes it
        a phaser rather than a pistol.

        AmmoType is base:bullets_9mm and that is deliberate, not laziness.
        AmmoTypes are registered in Java through AmmoType.registerBase and
        there is no script syntax anywhere in build 42 for a mod to add one --
        `bullets_9mm` appears in media/scripts only ever as the *value* of an
        AmmoType line, never as a definition. A made-up id resolves to nothing
        and the weapon will not fire, silently. Since the charge is restored
        far faster than it can be spent, no 9mm is ever drawn from the
        player's own stock; the ammunition type is a formality the engine
        insists on.

        MaxAmmo is high for the same reason: the top-up runs on a slow tick,
        and a deep charge means even a burst fired between two ticks cannot
        empty it.

        WeaponSprite names a vanilla in-hand model. A custom one needs a rigged
        attachment set rather than a static mesh, which is a different pipeline
        from the world models above; the inventory icon is the mod's own.  */

    item TrekPhaser
    {
        DisplayCategory = Weapon,
        ItemType = base:weapon,
        Type = Weapon,
        DisplayName = Phaser,
        Icon = TREK_Phaser,
        Weight = 0.6,

        Ranged = true,
        IsAimedFirearm = true,
        AmmoType = base:bullets_9mm,
        MaxAmmo = 60,
        ClipSize = 60,
        AmmoPerShoot = 1,
        WeaponReloadType = handgun,
        Reloadtime = 20,

        MinDamage = 1.2,
        MaxDamage = 1.9,
        CritDmgMultiplier = 5.0,
        CriticalChance = 40.0,
        DoorDamage = 12,
        MinRange = 0.61,
        MaxRange = 18.0,
        MinAngle = 0.92,
        MinSightRange = 2,
        MaxSightRange = 8.0,
        HitChance = 70,
        AimingPerkCritModifier = 6,
        AimingPerkHitChanceModifier = 4,
        AimingPerkMinAngleModifier = 0.05,
        AimingPerkRangeModifier = 0,
        Aimingtime = 18,
        RecoilDelay = 6,
        Swingtime = 0.5,
        MinimumSwingtime = 0.5,
        SwingAmountBeforeImpact = 0.0,
        SwingAnim = Handgun,
        RunAnim = Run_Weapon2,
        WeaponSprite = Handgun03,
        MaxHitcount = 1,
        Projectilecount = 1,
        PushBackMod = 0.3,
        KnockdownMod = 2.0,
        StopPower = 6.0,
        ToHitModifier = 1.5,
        MultipleHitConditionAffected = false,
        UseEndurance = false,

        /*  Never wears out and never jams. TREK_Phaser.lua puts the condition
            back anyway; setting these means it does not have to fight the
            engine to do it.  */
        ConditionMax = 100,
        ConditionLowerChanceOneIn = 10000,
        JamGunChance = 0,

        /*  Deliberately quiet. A phaser is not a gunshot, and a weapon with
            unlimited charge that also brought the whole street down on you
            would be no gift at all.  */
        SoundVolume = 8,
        SoundRadius = 12,
        SoundGain = 0.6,
        NPCSoundBoost = 1.0,
        SwingSound = TREK_PhaserPulse,
        HitSound = null,
        ImpactSound = null,
        EquipSound = M9Equip,
        UnequipSound = M9UnEquip,
        DropSound = M9Drop,
        BreakSound = null,
        ClickSound = null,
        RackSound = null,
        AimReleaseSound = null,
        BringToBearSound = null,
        InsertAmmoSound = null,
        InsertAmmoStartSound = null,
        InsertAmmoStopSound = null,
        EjectAmmoSound = null,
        EjectAmmoStartSound = null,
        EjectAmmoStopSound = null,
        ShellFallSound = null,

        SubCategory = Firearm,
        Tooltip = Tooltip_TREK_Phaser,
    }
}
